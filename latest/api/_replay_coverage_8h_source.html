11// The words of the replay row: which part of the run the replay holds, and why it is not all of
12// it. The window reads the history into a ReplayCoverage, asks for the text and writes it into
13// the two labels of the row and the tooltip of the Start button.
14//
15// The first text names the selected time as a clock time, when there is one. The second says which interval is
16// kept and how often, and last why the first part of the run is missing. It does not depend on the
17// selected time, so the row keeps its height while the user seeks. The interval is the one the
18// slider covers, and Start goes to its first time.

57inline ReplayCoverage replayCoverageOf(const GuiReplayHistory& history, long long clockOffsetSeconds) {
58 ReplayCoverage out;
59 if (!history.empty())

61 out.firstTime = history.firstTime();
62 out.lastTime = history.lastTime();
63 out.budgetBytes = history.budgetBytes();
64 out.clockOffsetSeconds = clockOffsetSeconds;

69inline std::string replayMemoryText(std::size_t bytes) {
70 const std::size_t mebibyte = 1024u * 1024u;
71 if (bytes % mebibyte == 0)
72 return std::to_string(bytes / mebibyte) + " MiB";
73 const std::size_t tenths = (bytes * 10 + mebibyte / 2) / mebibyte;
74 return std::to_string(tenths / 10) + "." + std::to_string(tenths % 10) + " MiB";
75}

79inline ReplayBarText replayBarText(const ReplayCoverage& coverage, std::optional<int> shownTime = std::nullopt) {
80 ReplayBarText out;
81 out.startTip = "Go to the start of the run";
82 const std::string memory = replayMemoryText(coverage.budgetBytes);
83 if (coverage.kind == ReplayCoverageKind::Unavailable) {
84 out.selected = "No replay";
85 out.coverage = "One moment of this run needs more than the replay memory (" + memory + ").";
86 return out;
87 }
88 out.usable = true;
89 out.selected = shownTime ? formatSimTime(*shownTime, coverage.clockOffsetSeconds) : "Replay";
90 const bool oneFrame = coverage.firstTime >= coverage.lastTime;
91 const std::string first = std::to_string(coverage.firstTime);
92 const std::string last = std::to_string(coverage.lastTime);
93 const std::string cadence = "a frame every " + std::to_string(coverage.cadenceSeconds) + " s";
94 if (coverage.kind == ReplayCoverageKind::WholeRun) {
95 out.coverage = "Whole run, " + (oneFrame ? "one frame at " + first + " s" : first + " to " + last + " s, " + cadence) + ".";
96 return out;
97 }
98 const std::string start = formatSimTime(coverage.firstTime, coverage.clockOffsetSeconds) + " (" + first + " s)";
99 out.coverage = (oneFrame ? "One frame at " + start : "Starts at " + start + ", ends at " + last + " s, " + cadence)
100 + ". The earlier part was not kept because the run is larger than the replay memory (" + memory + ").";
101 out.startTip = "Go to the first kept time, " + start + ". The earlier part of the run was not kept.";
102 return out;
103}
