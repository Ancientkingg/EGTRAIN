18inline bool copyTree(const QString& sourcePath, const QString& destinationPath) {
19 const QFileInfo source(sourcePath);
20 if (!source.isDir() || source.isSymLink())
21 return false;
22 if (!QDir().mkpath(destinationPath))
23 return false;
24 const QDir sourceDir(sourcePath);
25 const QFileInfoList entries = sourceDir.entryInfoList(
26 QDir::NoDotAndDotDot | QDir::AllEntries | QDir::Hidden | QDir::System,
27 QDir::DirsFirst | QDir::Name);
28 for (const QFileInfo& entry : entries) {
29 if (entry.isSymLink())
30 return false;
31 const QString destination = QDir(destinationPath).filePath(entry.fileName());
32 if (entry.isDir()) {
33 if (!copyTree(entry.absoluteFilePath(), destination))
34 return false;
35 continue;
36 }
37 if (!entry.isFile())
38 return false;
39 QFile::remove(destination);
40 if (!QFile::copy(entry.absoluteFilePath(), destination))
41 return false;
42 }
43 return true;
44}
