11// The words of the replay row: which part of the run the replay holds, and why it is not all of
12// it. The window reads the history into a ReplayCoverage, asks for the text and writes it into
13// the two labels of the row and the tooltip of the Start button.
14//
15// The first text names the selected time, when there is one. The second says which interval is
16// kept and how often, and last why the first part of the run is missing. It does not depend on the
17// selected time, so the row keeps its height while the user seeks. The interval is the one the
18// slider covers, and Start goes to its first time.
19
20enum class ReplayCoverageKind {
21 WholeRun, // every frame of the run is kept
22 FromLater, // the first part of the run was dropped to stay within the replay memory
23 Unavailable // not one frame fits in the replay memory
24};
25
26struct ReplayCoverage {
27 ReplayCoverageKind kind = ReplayCoverageKind::Unavailable;
28 // First and last time that is kept, in seconds since simulation start.
29 int firstTime = 0;
30 int lastTime = 0;
31 int cadenceSeconds = GuiReplayHistory::cadenceSeconds;
32 // Size of the replay memory in bytes.
33 std::size_t budgetBytes = GuiReplayHistory::payloadLimit;
34 // Simulation start, seconds since midnight (the offset formatSimTime takes).
35 long long clockOffsetSeconds = 0;
36};

38struct ReplayBarText {
39 // For the label beside the slider: "Replay at 1230 s", "Replay" while no frame is selected, or "No replay".
40 std::string selected;
41 // For the label below the controls: the kept interval and why the rest of the run is missing.
42 std::string coverage;
43 // Tooltip of the Start button. It is empty while Start goes to the start of the run.

49// A history without frames cannot be replayed, whether it was never filled or one frame was too
50// large for it.
51inline ReplayCoverage replayCoverageOf(const GuiReplayHistory& history, long long clockOffsetSeconds) {
52 ReplayCoverage out;
53 if (!history.empty())
54 out.kind = history.truncated() ? ReplayCoverageKind::FromLater : ReplayCoverageKind::WholeRun;
55 out.firstTime = history.firstTime();
56 out.lastTime = history.lastTime();
57 out.budgetBytes = history.budgetBytes();
58 out.clockOffsetSeconds = clockOffsetSeconds;
59 return out;
60}

62// The size of the replay memory: whole mebibytes as "128 MiB", otherwise one decimal as "2.5 MiB".
63inline std::string replayMemoryText(std::size_t bytes) {
64 const std::size_t mebibyte = 1024u * 1024u;
65 if (bytes % mebibyte == 0)
66 return std::to_string(bytes / mebibyte) + " MiB";
67 const std::size_t tenths = (bytes * 10 + mebibyte / 2) / mebibyte;
68 return std::to_string(tenths / 10) + "." + std::to_string(tenths % 10) + " MiB";
69}

71// shownTime is the time of the frame on the canvas. It is left out while no frame has been
72// selected yet, as when the run has just finished.
73inline ReplayBarText replayBarText(const ReplayCoverage& coverage, std::optional<int> shownTime = std::nullopt) {
74 ReplayBarText out;
75 const std::string memory = replayMemoryText(coverage.budgetBytes);
76 if (coverage.kind == ReplayCoverageKind::Unavailable) {
77 out.selected = "No replay";
78 out.coverage = "One moment of this run needs more than the replay memory (" + memory + ").";
79 return out;
80 }
81 out.usable = true;
82 out.selected = shownTime ? "Replay at " + std::to_string(*shownTime) + " s" : "Replay";
83 const bool oneFrame = coverage.firstTime >= coverage.lastTime;
84 const std::string first = std::to_string(coverage.firstTime);
85 const std::string last = std::to_string(coverage.lastTime);
86 const std::string cadence = "a frame every " + std::to_string(coverage.cadenceSeconds) + " s";
87 if (coverage.kind == ReplayCoverageKind::WholeRun) {
88 out.coverage = "Whole run, " + (oneFrame ? "one frame at " + first + " s" : first + " to " + last + " s, " + cadence) + ".";
89 return out;
90 }
91 const std::string start = formatSimTime(coverage.firstTime, coverage.clockOffsetSeconds) + " (" + first + " s)";
92 out.coverage = (oneFrame ? "One frame at " + start : "Starts at " + start + ", ends at " + last + " s, " + cadence)
93 + ". The earlier part was not kept because the run is larger than the replay memory (" + memory + ").";
94 out.startTip = "Go to the first kept time, " + start + ". The earlier part of the run was not kept.";
95 return out;
96}
