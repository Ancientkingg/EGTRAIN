8// Occupation of a set of resources by a set of trains.
9//
10// Every time is in seconds, every percentage is in percent (the unit of
11// CapacityAnalysisResult::cyclePercentage) and -1.0 marks a value that is not
12// available (the sentinel of CapacityAnalysisResult).
13//
14// The analysis area is only the list AreaCapacityInput::resourceIds of one
15// call. It is not a signalling area, reads no signalling level and is stored
16// nowhere. A resource is a block id as BlockingTimeDiagramInput::blockId holds
17// it. An occupation belongs to a resource when shareBlockingTimeResource is
18// true for the two ids. A composite block id therefore counts for every
19// resource it shares a component with (once in each such row and once in the
20// envelope), so the caller lists plain block ids. A resource id that is itself
21// a composite id matches in the same way, so two rows can count the same
22// occupation. Occupations are not clipped to a time window: the caller chooses
23// the trains of the period and the period is only the denominator.
24
25struct AreaCapacityTrain {
26 CapacityAnalysisTrain train;
27 // Route::reversed_direction of the route of the train. The chain keeps the
28 // trains that have the value of the first selected train.
29 bool reversedDirection = false;
30};

32struct AreaCapacityInput {

45// One row per resource of the area, computed from the usable occupations of
46// the selected trains that belong to the resource.
47struct AreaCapacityResource {
48 std::string resourceId;
49 // Selected trains with at least one such occupation.
50 int trainCount = 0;
51 // Such occupations; a repeat on the same resource counts again.
52 int occupationCount = 0;
53 // Sum of endOccTime minus startOccTime over those occupations.
54 double occupationSeconds = 0.0;
55 // Length of the union of those occupations; overlapping or touching ones count once.
56 double unionSeconds = 0.0;
57 // occupationSeconds / period * 100; -1.0 without a valid period.
58 double occupationPercentage = -1.0;
59 // True for the rows whose occupationSeconds is the largest (within
60 // kBlockingTimeToleranceSeconds, ties all count) and above 0.
61 bool governing = false;
62};
