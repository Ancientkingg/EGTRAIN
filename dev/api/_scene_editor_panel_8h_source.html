23inline int rowOfEntry(const QListWidget* list, const QString& text, int row) {
24 if (text.isEmpty())

38inline void restoreListSelection(QListWidget* list, const QString& text, int row) {
39 const int count = list->count();
40 const int sameEntry = rowOfEntry(list, text, row);
41 if (sameEntry >= 0)
42 list->setCurrentRow(sameEntry);
43 else
44 list->setCurrentRow(count == 0 ? -1 : (row < 0 ? 0 : std::min(row, count - 1)));
