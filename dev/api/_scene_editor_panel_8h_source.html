10// The row of the entry that reads `text` in a list that was filled again, or -1 when no entry does. The entry at `row` is
11// preferred when it reads `text`, so that entries that share their text keep their place. A panel uses it to keep the
12// selected entry selected when the document is replaced by one that lists the entry at another row.
13inline int rowOfEntry(const QListWidget* list, const QString& text, int row) {
14 if (text.isEmpty())
15 return -1;
16 if (row >= 0 && row < list->count() && list->item(row)->text() == text)
17 return row;
18 for (int index = 0; index < list->count(); ++index) {
19 if (list->item(index)->text() == text)
20 return index;
21 }
22 return -1;
23}
