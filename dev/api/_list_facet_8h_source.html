18// A facet over one list of the model, whose items have an id and are listed in the order of the list. The column of the

35 void commitCell(InfrastructurePanel& panel, int row, int column) override {
36 std::vector<Item>& items = panel.mutableModel().*m_list;
37 if (row >= static_cast<int>(items.size()))
38 return;
39 const QString value = panel.table()->item(row, column)->text().trimmed();
40 Item& item = items[static_cast<std::size_t>(row)];
41 Edit edit = Edit::Unchanged;
42 if (column == 0) {
43 const std::string newId = value.toStdString();
44 if (item.id != newId) {
45 if (idCanReplaceRow(items, row, newId)) {
46 const std::string oldId = item.id;
47 item.id = newId;
48 rewriteReferences(panel.mutableModel(), oldId, newId);
49 panel.setSelectionId(value);
50 edit = Edit::Changed;
51 } else {
52 edit = Edit::UnsafeId;
53 }
54 }
55 } else {
56 edit = editField(item, column, value);
57 }
58 if (edit == Edit::Changed)
59 panel.entityChanged((column == 0 ? QStringLiteral("Rename %1") : QStringLiteral("Edit %1")).arg(m_noun));
60 else if (edit == Edit::UnsafeId)
61 panel.refuseEdit(QStringLiteral("Infrastructure IDs must be non-empty and unique"));
62 else if (edit == Edit::Refused)
63 panel.refuseEdit();
64 }

74 void remove(InfrastructurePanel& panel) override {
75 std::vector<Item>& items = panel.mutableModel().*m_list;
76 const int row = panel.table()->currentRow();
77 if (row < 0 || row >= static_cast<int>(items.size()))
78 return;
79 const std::string id = items[static_cast<std::size_t>(row)].id;
80 if (id.empty())
81 return;

83 return;
84 items.erase(items.begin() + row);
