158 SignalState state; // Main signal aspect
159 bool Occupied; // Block section occupied by a train
160 bool Occup_By_Train; // Distinguishes train-on-section (true) from switch-occupied (false)
161 bool withSwitchDiv; // Block section contains a diverging switch

163 int N_ConnectedBS; // Number of connected block sections

166 Section();
167
168 Section& operator=(const Section& ob2) {
169 if (this == &ob2)
170 return *this;
171 ID = ob2.ID;
172 start_node = ob2.start_node;
173 end_node = ob2.end_node;
174 total_nodes = ob2.total_nodes;
175 total_arcs = ob2.total_arcs;
176 GeoXBegNode = ob2.GeoXBegNode;
177 GeoXEndNode = ob2.GeoXEndNode;
178 length = ob2.length;
179 exit_speed = ob2.exit_speed;
180 code = ob2.code;
181 SignallingLevel = ob2.SignallingLevel;
182 nodelist_of_nodes_in_signalling_section = ob2.nodelist_of_nodes_in_signalling_section;
183 FirstConnectedTrackLineID = ob2.FirstConnectedTrackLineID;
184 SecondConnectedTrackLineID = ob2.SecondConnectedTrackLineID;
185
186 for (int i = 0; i < total_arcs; i++)
187 arcs_in_signalling_block_section[i] = ob2.arcs_in_signalling_block_section[i];
188
189 state = ob2.state;
190 trackLineId = ob2.trackLineId;
191 withSwitchDiv = ob2.withSwitchDiv;
192 Occupied = ob2.Occupied;
193 Occup_By_Train = ob2.Occup_By_Train;
194 N_ConnectedBS = ob2.N_ConnectedBS;
195 XStartSwitch = ob2.XStartSwitch;
196 XEndSwitch = ob2.XEndSwitch;
197 for (int i = 0; i < N_ConnectedBS; i++) {
198 IDConnectedBS[i] = ob2.IDConnectedBS[i];
199 }
200 // Copy ETCS3 braking points
201 N_ETCS3BrakingPoints = ob2.N_ETCS3BrakingPoints;
202 for (int i = 0; i < N_ETCS3BrakingPoints; i++) {
203 ETCS3BrakingPoints[i] = ob2.ETCS3BrakingPoints[i];
204 ETCS3BrakingPointsTrainID[i] = ob2.ETCS3BrakingPointsTrainID[i];
205 }
206 TDS_in_block.clear();
207
208 if (ob2.TDS_in_block.size() > 0) {
209 for (auto k = ob2.TDS_in_block.begin(); k != ob2.TDS_in_block.end(); k++) {
210 this->TDS_in_block.push_back(*k);
211 }
212 }
213
214 return *this;
215 }
216
217 // Equality operator for block sections
218 bool operator==(const Section& ob2) {
219 bool AreArcsEqual = true;
220 for (int i = 0; i < total_arcs && i < ob2.total_arcs; i++) {
221 if (!(arcs_in_signalling_block_section[i] == ob2.arcs_in_signalling_block_section[i])) {
222 AreArcsEqual = false;
223 break;
224 }
225 }
226 if ((ID == ob2.ID) && (total_nodes == ob2.total_nodes)
227 && (total_arcs == ob2.total_arcs)
228 && (length == ob2.length)
229 && (start_node == ob2.start_node)
230 && (nodelist_of_nodes_in_signalling_section == ob2.nodelist_of_nodes_in_signalling_section)
231 && AreArcsEqual
232 && (exit_speed == ob2.exit_speed)
233 && (code == ob2.code)
234 && (state == ob2.state))
235 return true;
236 else
237 return false;
238 }
