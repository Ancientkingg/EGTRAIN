73inline std::error_code uniqueSiblingPath(const std::filesystem::path& parent, const std::string& prefix,
74 std::filesystem::path& result) {
75 std::error_code ec;
76 for (unsigned int attempt = 0; attempt < 100; ++attempt) {
77 const auto stamp = std::chrono::steady_clock::now().time_since_epoch().count();
78 const std::filesystem::path candidate = parent / pathFromUtf8(prefix + std::to_string(stamp) + "-" + std::to_string(attempt));
79 ec.clear();
80 const auto status = std::filesystem::symlink_status(candidate, ec);
81 if (!ec && status.type() == std::filesystem::file_type::not_found) {
82 result = candidate;
83 return std::error_code();
84 }
85 if (ec == std::errc::no_such_file_or_directory) {
86 result = candidate;
87 return std::error_code();
88 }
89 if (ec)
90 return ec;
91 }
92 return std::make_error_code(std::errc::file_exists);
93}

98inline std::error_code createStagedDirectory(const std::filesystem::path& parent, const std::string& prefix,
99 bool ownerOnly, StagedDirectory& staged) {
100 std::error_code ec;
101 for (unsigned int attempt = 0; attempt < 100; ++attempt) {
102 const auto stamp = std::chrono::steady_clock::now().time_since_epoch().count();
103 const std::filesystem::path candidate = parent / pathFromUtf8(prefix + std::to_string(stamp) + "-" + std::to_string(attempt));
104 ec.clear();
105 if (std::filesystem::create_directory(candidate, ec)) {
106 if (ownerOnly) {
107 ec.clear();
108 std::filesystem::permissions(candidate, std::filesystem::perms::owner_all,
109 std::filesystem::perm_options::replace, ec);
110 if (ec) {
111 std::error_code cleanupError;
112 std::filesystem::remove_all(candidate, cleanupError);
113 return ec;
114 }
115 }
116 staged.path = candidate;
117 return std::error_code();
118 }
119 if (ec && ec != std::errc::file_exists)
120 return ec;
121 }
122 return std::make_error_code(std::errc::file_exists);
123}
