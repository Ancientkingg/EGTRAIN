68inline std::error_code uniqueSiblingPath(const std::filesystem::path& parent, const std::string& prefix,
69 std::filesystem::path& result) {
70 std::error_code ec;
71 for (unsigned int attempt = 0; attempt < 100; ++attempt) {
72 const auto stamp = std::chrono::steady_clock::now().time_since_epoch().count();
73 const std::filesystem::path candidate = parent / pathFromUtf8(prefix + std::to_string(stamp) + "-" + std::to_string(attempt));
74 ec.clear();
75 const auto status = std::filesystem::symlink_status(candidate, ec);
76 if (!ec && status.type() == std::filesystem::file_type::not_found) {
77 result = candidate;
78 return std::error_code();
79 }
80 if (ec == std::errc::no_such_file_or_directory) {
81 result = candidate;
82 return std::error_code();
83 }
84 if (ec)
85 return ec;
86 }
87 return std::make_error_code(std::errc::file_exists);

93inline std::error_code createStagedDirectory(const std::filesystem::path& parent, const std::string& prefix,
94 bool ownerOnly, StagedDirectory& staged) {
95 std::error_code ec;
96 for (unsigned int attempt = 0; attempt < 100; ++attempt) {
97 const auto stamp = std::chrono::steady_clock::now().time_since_epoch().count();
98 const std::filesystem::path candidate = parent / pathFromUtf8(prefix + std::to_string(stamp) + "-" + std::to_string(attempt));
99 ec.clear();
100 if (std::filesystem::create_directory(candidate, ec)) {
101 if (ownerOnly) {
102 ec.clear();

104 std::filesystem::perm_options::replace, ec);
105 if (ec) {
106 std::error_code cleanupError;
107 std::filesystem::remove_all(candidate, cleanupError);
108 return ec;
109 }
110 }
111 staged.path = candidate;
112 return std::error_code();
113 }
114 if (ec && ec != std::errc::file_exists)
115 return ec;
116 }
117 return std::make_error_code(std::errc::file_exists);

139inline StagedPublishResult publishStagedDirectory(StagedDirectory& staged, const std::filesystem::path& destination,
140 const std::filesystem::path& backup) {
141 StagedPublishResult result;
142 if (!backup.empty()) {
143 std::filesystem::rename(destination, backup, result.error);
144 if (result.error) {
145 result.step = StagedPublishStep::MoveToBackup;
146 return result;
147 }
148 }
149 std::filesystem::rename(staged.path, destination, result.error);
150 if (result.error) {
151 result.step = StagedPublishStep::Rename;
152 if (!backup.empty())
153 std::filesystem::rename(backup, destination, result.restoreError);
154 return result;
155 }
156 staged.release();
157 if (!backup.empty())
158 std::filesystem::remove_all(backup, result.removeBackupError);
159 return result;
160}
