69// Keeps recovery copies of a scene below a root folder that the caller names. Each running
70// application (an instance) owns one folder <root>/<instance id>, holds a lock on it while the
71// store is open and keeps at most one copy of the scene in it, as an ordinary scene bundle. A
72// clean close removes the folder. A folder whose owner is gone is a candidate for recovery.
73//
74// The store writes only below its root, apart from the private copy that saveSceneBundle makes
75// in the system temporary directory while it runs. It never touches the file or folder a scene
76// came from. No function throws, prints or logs, and no error text holds a path.
77//
78// A store is used by one thread at a time and has no static state.
79class RecoveryStore {
80public:
81 // At most this many candidates are kept; scan removes the oldest beyond it.
82 static constexpr int kMaxKeptInstances = 20;
83
84 explicit RecoveryStore(const QString& root);
85 // Releases the lock and deletes nothing, so a missing close() never destroys a copy.
86 ~RecoveryStore();
87 RecoveryStore(const RecoveryStore&) = delete;
88 RecoveryStore& operator=(const RecoveryStore&) = delete;
89
90 // Creates the instance folder with a new random id and locks it. Ok when already open.
91 RecoveryResult open();
92 bool isOpen() const;
93 QString instanceId() const;
94
95 // Writes the scene as a new generation next to the previous one, then records it and removes
96 // the previous one. A failed write leaves the previous copy.
97 RecoveryResult write(const SceneModel& scene, const RecoverySource& source);
98 // Removes the copy and the record. The folder and the lock stay.
99 RecoveryResult clear();
100 // Removes the instance folder and releases the lock. Ok when already closed.
101 RecoveryResult close();
102
103 // Lists the copies that other applications left behind, newest first. A folder whose lock is
104 // held belongs to a running application and is skipped. The scan removes what is not usable:
105 // stray files of an interrupted write, older generations, folders without a complete bundle,
106 // and candidates beyond kMaxKeptInstances.
107 RecoveryScan scan();
108 // Takes over the copy of an Openable candidate. It replaces the copy this store holds, as write
109 // does, and the candidate folder is removed. Fails when the candidate is in use.
110 RecoveryResult adopt(const RecoveryCandidate& candidate);
111 // Removes the folder of a candidate in any state. Fails when the candidate is in use.
112 RecoveryResult discard(const RecoveryCandidate& candidate);
113
114private:
115 QString m_root;
116 QString m_id;
117 std::unique_ptr<QLockFile> m_lock;
118 int m_generation = 0; // Number of the newest generation this store wrote.
119};
