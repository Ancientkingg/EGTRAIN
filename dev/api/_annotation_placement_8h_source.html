17inline bool annotationSignalNormal(const TrackPreviewLine& line, double rawX,
18 std::pair<double, double>& normal) {
19 if (!std::isfinite(rawX))
20 return false;
21 bool foundEndpoint = false;
22 for (std::size_t i = 1; i < line.points.size(); ++i) {
23 const auto& a = line.points[i - 1];
24 const auto& b = line.points[i];
25 if (!std::isfinite(a.rawX) || !std::isfinite(b.rawX)
26 || !std::isfinite(a.x) || !std::isfinite(a.y)
27 || !std::isfinite(b.x) || !std::isfinite(b.y)
28 || a.rawX == b.rawX || rawX < std::min(a.rawX, b.rawX)
29 || rawX > std::max(a.rawX, b.rawX))
30 continue;
31 const double direction = b.rawX > a.rawX ? 1.0 : -1.0;
32 const double dx = (b.x - a.x) * direction;
33 const double dy = (b.y - a.y) * direction;
34 const double length = std::hypot(dx, dy);
35 if (!std::isfinite(length) || length == 0.0)
36 continue;
37 normal = {dy / length, -dx / length};
38 // Historical intervals are right-open, except at the final endpoint.
39 // Prefer the outgoing interval at a bend regardless of storage order.
40 if (rawX < std::max(a.rawX, b.rawX))
41 return true;
42 foundEndpoint = true;
43 }
44 return foundEndpoint;
45}

141inline std::pair<double, double> annotationStationShift(
142 const std::vector<SceneStationView>& views, const std::string& stationId) {
143 struct Position {
144 double chainage;
145 std::string id;
146 double x;
147 double y;
148 };
149 std::map<int, std::vector<Position>> regions;
150 std::vector<int> stationRegions;
151 constexpr double pi = 3.14159265358979323846;
152 for (const auto& view : views) {
153 if (!std::isfinite(view.latitude) || std::abs(view.latitude) >= 90.0
154 || !std::isfinite(view.longitude))
155 continue;
156 const double y = -std::log(std::tan(pi / 4.0 + view.latitude * pi / 360.0)) * 180.0 / pi;
157 if (!std::isfinite(y))
158 continue;
159 for (const auto& region : view.regions) {
160 if (!std::isfinite(region.second))
161 continue;
162 regions[region.first].push_back({region.second, view.stationId, view.longitude, y});
163 if (view.stationId == stationId
164 && std::find(stationRegions.begin(), stationRegions.end(), region.first) == stationRegions.end())
165 stationRegions.push_back(region.first);
166 }
167 }
168 double sumX = 0.0, sumY = 0.0;
169 std::size_t count = 0;
170 for (int region : stationRegions) {
171 auto& positions = regions[region];
172 std::sort(positions.begin(), positions.end(), [](const Position& a, const Position& b) {
173 return a.chainage < b.chainage;
174 });
175 if (stationId.empty()
176 || std::adjacent_find(positions.begin(), positions.end(),
177 [](const Position& a, const Position& b) { return a.chainage == b.chainage; })
178 != positions.end())
179 continue;
180 const auto target = std::find_if(positions.begin(), positions.end(),
181 [&stationId](const Position& p) { return p.id == stationId; });
182 if (target == positions.end() || positions.size() < 2)
183 continue;
184 const std::size_t index = static_cast<std::size_t>(target - positions.begin());
185 const std::size_t middle = index == positions.size() - 1 ? index - 1 : index;
186 const Position& a = positions[middle == 0 ? 0 : middle - 1];
187 const Position& b = positions[middle == 0 ? 1 : middle];
188 const Position& c = positions[std::min(middle == 0 ? 2 : middle + 1, positions.size() - 1)];
189 double sx, sy;
190 if (positions.size() == 2) {
191 const double dx = positions[1].x - positions[0].x;
192 const double dy = positions[1].y - positions[0].y;
193 const double length = std::hypot(dx, dy);
194 if (!std::isfinite(length) || length == 0.0)
195 continue;
196 sx = dy / length;
197 sy = -dx / length;
198 } else {
199 const double ux = a.x - b.x, uy = a.y - b.y;
200 const double vx = c.x - b.x, vy = c.y - b.y;
201 if (!std::isfinite(std::hypot(ux, uy)) || !std::isfinite(std::hypot(vx, vy))
202 || std::hypot(ux, uy) == 0.0 || std::hypot(vx, vy) == 0.0)
203 continue;
204 double alpha = std::atan2(ux * vy - uy * vx, ux * vx + uy * vy);
205 if (alpha < 0.0)
206 alpha += 2.0 * pi;
207 const double beta = std::atan2(-uy, ux);
208 const double gamma = alpha / 2.0 - beta;
209 sx = std::cos(gamma);
210 sy = std::sin(gamma);
211 }
212 if (!std::isfinite(sx) || !std::isfinite(sy))
213 continue;
214 sumX += sx;
215 sumY += sy;
216 ++count;
217 }
218 if (count == 0)
219 return {0.0, 0.0};
220 return {-1200.0 * sumX / count, -1200.0 * sumY / count};
221}
