31// Reads a JSON integer into an int. Returns false, leaving output unchanged, when the value is not
32// an integer or lies outside [min, max].
33inline bool readJsonInt(const nlohmann::json& value, int min, int max, int& output) {
34 long long number = 0;
35 if (value.is_number_unsigned()) {
36 const unsigned long long unsignedNumber = value.get<unsigned long long>();
37 if (unsignedNumber > static_cast<unsigned long long>(LLONG_MAX))
38 return false;
39 number = static_cast<long long>(unsignedNumber);
40 } else if (value.is_number_integer()) {
41 number = value.get<long long>();
42 } else {
43 return false;
44 }
45 if (number < min || number > max)
46 return false;
47 output = static_cast<int>(number);
48 return true;
49}
