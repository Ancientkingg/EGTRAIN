11// The codes and message wording of the errors that a SceneFieldReader reports. In the section messages % stands for the
12// key. The scene files and the standalone scenario file each have their own set.
13struct SceneFieldCodes {
14 const char* itemInvalid; // code: an item is not an object
15 const char* itemMessage; // message for an item that a field is read from

46// Reads typed fields from the JSON of one scene file and appends an error to the sink for every field that is absent or
47// invalid. Each reader returns whether it stored a value; an optional field that is absent is not an error. The file name
48// and the sink must outlive the reader.
49class SceneFieldReader {
50public:
51 SceneFieldReader(std::string_view file, const SceneFieldCodes& codes, std::vector<SceneDiagnostic>& sink)
52 : file_(file), codes_(codes), sink_(sink) {}
53
54 // Reports item and returns false when it is not an object. arraySection and the field readers (except boolField) do
55 // this themselves for the object that they read from, once per call; call it to report an item once, with its own
56 // message, before its fields are read.
57 bool objectItem(const nlohmann::json& item, const std::string& path, const char* message) const;
58 bool arraySection(const nlohmann::json& object, const char* key, const std::string& path, bool required) const;
59 bool stringField(const nlohmann::json& object, const char* key, const std::string& path, std::string& output,
60 bool required = true) const;
61 bool numberField(const nlohmann::json& object, const char* key, const std::string& path, double& output,
62 bool required = true) const;
63 // A value that does not fit an int is invalid.
64 bool integerField(const nlohmann::json& object, const char* key, const std::string& path, int& output,
65 bool required = true) const;
66 // Always optional. Reports nothing for an absent key or for an item that is not an object.
67 bool boolField(const nlohmann::json& object, const char* key, const std::string& path, bool& output) const;
68 // Reports an error with its own code, for a check that is not about the type of one field.
69 void report(const char* code, const std::string& message, const std::string& path) const;
70
71private:
72 std::string_view file_;
73 const SceneFieldCodes& codes_;
74 std::vector<SceneDiagnostic>& sink_;
75};
