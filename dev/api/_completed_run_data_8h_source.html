32// the train's own series ended, so a sample the train did not record is absent, not zero.

36 int firstSample = 0;
37 std::vector<double> position; // m
38 std::vector<double> speed; // m/s
39 std::vector<double> power; // W
40 std::vector<double> energy; // MJ
41 std::vector<double> effort; // N
42 std::vector<std::uint32_t> block; // Index into CompletedRunData::blockNames, or kNoBlock

44
45 std::optional<double> at(const std::vector<double>& series, int sample) const {
46 const long long offset = static_cast<long long>(sample) - firstSample;
47 if (offset < 0 || offset >= static_cast<long long>(series.size()))
48 return std::nullopt;
49 return series[static_cast<std::size_t>(offset)];
50 }
51 std::uint32_t blockAt(int sample) const {
52 const long long offset = static_cast<long long>(sample) - firstSample;
53 return offset < 0 || offset >= static_cast<long long>(block.size()) ? kNoBlock : block[static_cast<std::size_t>(offset)];
54 }
55};

70 return !routeBlockIds.empty();
71 return std::any_of(routeBlockIds.begin(), routeBlockIds.end(), [&allowedBlockIds](const std::string& id) {
