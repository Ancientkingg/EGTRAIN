11// Lets the mouse wheel change a combo box, spin box or slider only while it has keyboard focus. Otherwise the wheel
12// passes to the parent, so that it scrolls a form instead of changing a value on the way. The box takes focus
13// from a click or the tab key but not from the wheel.
14namespace FocusWheel {

16class Filter : public QObject {

18 using QObject::QObject;

20protected:
21 bool eventFilter(QObject* watched, QEvent* event) override {
22 auto* box = qobject_cast<QWidget*>(watched);
23 if (event->type() != QEvent::Wheel || !box || box->hasFocus())

30inline void guard(QWidget* box) {
31 box->setFocusPolicy(Qt::StrongFocus);
32 // A box that guards itself is visited again by guardAll; the property keeps it from getting a second filter.
33 if (box->property("focusWheelGuard").toBool())
34 return;
35 box->setProperty("focusWheelGuard", true);

39// Whether the wheel changes the value of widget. A scroll bar is not one of them: it has to follow the wheel.
40inline bool isValueField(const QWidget* widget) {
41 return qobject_cast<const QComboBox*>(widget) || qobject_cast<const QAbstractSpinBox*>(widget) || qobject_cast<const QSlider*>(widget);
42}

44// Guards every combo box, spin box and slider inside root.
45inline void guardAll(QWidget* root) {
46 for (auto* field : root->findChildren<QWidget*>()) {
47 if (isValueField(field))

52// Keeps every field of root guarded, also the fields that a table, a tab or a later refresh adds after the call. A new
53// widget is guarded when it is polished, before it can be shown and meet the wheel.
54class Watcher : public QObject {
55public:
56 explicit Watcher(QWidget* root) : QObject(root) {
57 add(root);
58 }
59
60protected:
61 bool eventFilter(QObject* watched, QEvent* event) override {

63 if (auto* child = qobject_cast<QWidget*>(static_cast<QChildEvent*>(event)->child()))
64 add(child);
65 }
66 return QObject::eventFilter(watched, event);
67 }

69private:
70 void add(QWidget* widget) {
71 if (isValueField(widget))

73 if (widget->property("focusWheelWatched").toBool())
74 return;
75 widget->setProperty("focusWheelWatched", true);
76 widget->installEventFilter(this);
