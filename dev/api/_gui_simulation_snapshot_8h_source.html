132class GuiSignalStateList {
133public:
136 void merge(std::string_view id, bool reversed, int code, int level) {
137 if (!guiSignalHasLevel(level))
138 level = kGuiSignalNoLevel;
139 const std::string key(id);
140 auto found = index_.find(key);
141 if (found == index_.end())
142 found = index_.emplace(key, std::array<int, 2>{-1, -1}).first;
143 int& position = found->second[reversed ? 1 : 0];
144 if (position < 0) {
145 position = static_cast<int>(states_.size());
146 states_.push_back({key, code, reversed, level, false});
147 return;
148 }
149 GuiSignalState& into = states_[static_cast<std::size_t>(position)];
150 if (!guiSignalHasLevel(level))
151 return;
152 if (!guiSignalHasLevel(into.level)) {
153 into.code = code;
154 into.level = level;
155 return;
156 }
157 if (guiSignalRestriction(code) < guiSignalRestriction(into.code))
158 into.code = code;
159 into.level = std::min(into.level, level);
160 }
161
163 void fail(std::string_view id) {
164 const auto found = index_.find(std::string(id));
165 if (found == index_.end())
166 return;
167 for (const int position : found->second)
168 if (position >= 0)
169 states_[static_cast<std::size_t>(position)].failed = true;
170 }
171
173 std::vector<GuiSignalState> take() {
174 index_.clear();
175 return std::move(states_);
176 }
177
178private:
179 std::vector<GuiSignalState> states_;
180 // Position in states_ per direction (forward, reversed), or -1.
181 std::unordered_map<std::string, std::array<int, 2>> index_;
182};
