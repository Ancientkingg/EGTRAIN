25// Label for data and sentences in an editor panel, such as a path, an id or an explanation. The
26// text is plain and wraps at spaces. Layouts ignore the width hints of the label, so a word wider
27// than the width its layout gives it is clipped at the right edge instead of widening the panel.
28// The label has no role and stays visible when its text is empty. The caller adds it to a layout.
29inline QLabel* wrappedLabel(const QString& text, QWidget* parent) {
30 auto* label = new QLabel(parent);
31 label->setTextFormat(Qt::PlainText);
32 label->setWordWrap(true);
33 QSizePolicy policy = label->sizePolicy();
34 policy.setHorizontalPolicy(QSizePolicy::Ignored);
35 label->setSizePolicy(policy);
36 label->setText(text);
37 return label;
38}

50// Heading for a group of fields: bold and two points larger than the font of the parent when the
51// heading is made, or three pixels larger when that font is given in pixels. The text is plain and
52// wraps as in wrappedLabel. The caller adds it to a layout and sets any object name.
53inline QLabel* sectionHeading(const QString& title, QWidget* parent) {
54 QLabel* heading = wrappedLabel(title, parent);
55 QFont font = heading->font();
56 font.setBold(true);
57 if (font.pointSizeF() > 0)
58 font.setPointSizeF(font.pointSizeF() + 2.0);
59 else if (font.pixelSize() > 0)
60 font.setPixelSize(font.pixelSize() + 3);
61 heading->setFont(font);
62 return heading;
63}

65// The name with a zero-width space after each underscore. A wrapped label breaks a line at a space only, so
66// a name such as Amsterdam_Hilversum_Student would be clipped at the edge of its panel. With the added characters
67// it wraps between its parts. The characters are not drawn.
68inline QString breakableName(QString name) {
69 return name.replace(QLatin1Char('_'), QString(QLatin1Char('_')) + QChar(0x200B));
70}

72// Scroll area for the panel of a narrow rail dock. It scrolls up and down and never shows a horizontal scroll
73// bar. Its minimum width is the minimum width of the panel plus the vertical scroll bar, so the dock cannot be
74// made narrower than the panel needs and the scroll bar does not take room from the fields of the panel. The
75// panel is resized with the area and has no frame.
76class VerticalScrollArea : public QScrollArea {
77public:
78 explicit VerticalScrollArea(QWidget* parent = nullptr)
79 : QScrollArea(parent) {
80 setWidgetResizable(true);
81 setFrameShape(QFrame::NoFrame);
82 setHorizontalScrollBarPolicy(Qt::ScrollBarAlwaysOff);
83 }
84
85 QSize minimumSizeHint() const override {
86 QSize hint = QScrollArea::minimumSizeHint();
87 if (const QWidget* panel = widget())
88 hint.setWidth(panel->minimumSizeHint().width() + verticalScrollBar()->sizeHint().width() + 2 * frameWidth());
89 return hint;
90 }
91};

93// Scroll area for the panel of an editor dock. QScrollArea keeps the size hint of the panel from the first time it was
94// asked, which can be before the panel shows its tabs and tables, so the hint of this area is at least the width that the
95// panel needs now, with the vertical scroll bar. As in QScrollArea the hint is at most 36 line heights wide.
96// While the dock is docked in a main window, the hint is also at most five eighths of the width that the docks on the left
97// of the window leave, but not under 12 line heights. A dock with large text then opens narrower than its panel, which
98// scrolls inside it, and the network canvas keeps the rest.
99class DockScrollArea : public QScrollArea {
100public:
101 explicit DockScrollArea(QWidget* parent = nullptr)
102 : QScrollArea(parent) {}
103
104 QSize sizeHint() const override {
105 QSize hint = QScrollArea::sizeHint();
106 const QWidget* panel = widget();
107 if (!panel)
108 return hint;
109 const int lineHeight = fontMetrics().height();
110 const int needed = panel->minimumSizeHint().width() + verticalScrollBar()->sizeHint().width() + 2 * frameWidth();
111 hint.setWidth(std::min(std::max(hint.width(), needed), 36 * lineHeight));
112 hint.setWidth(std::min(hint.width(), std::max(widthLimitInWindow(), 12 * lineHeight)));
113 return hint;
114 }
115
116 // A change of the layout of the panel also changes the size hint of this area, which a dock keeps until told.
117 bool eventFilter(QObject* watched, QEvent* event) override {
118 if (watched == widget() && event->type() == QEvent::LayoutRequest)
119 updateGeometry();
120 return QScrollArea::eventFilter(watched, event);
121 }
122
123private:
124 // The widest that the dock opens in its main window; unlimited while the dock floats.
125 int widthLimitInWindow() const {
126 const auto* dock = qobject_cast<const QDockWidget*>(parentWidget());
127 const auto* window = dock && !dock->isFloating() ? qobject_cast<const QMainWindow*>(dock->parentWidget()) : nullptr;
128 if (!window)
129 return std::numeric_limits<int>::max();
130 int beside = window->width();
131 for (QDockWidget* other : window->findChildren<QDockWidget*>(QString(), Qt::FindDirectChildrenOnly)) {
132 if (other != dock && other->isVisible() && !other->isFloating() && window->dockWidgetArea(other) == Qt::LeftDockWidgetArea)
133 beside -= other->width();
134 }
135 return beside * 5 / 8;
136 }
137};
