128private:
129 // The widest that the dock opens in its main window; unlimited while the dock floats.
130 int widthLimitInWindow() const {
131 const auto* dock = qobject_cast<const QDockWidget*>(parentWidget());
132 const auto* window = dock && !dock->isFloating() ? qobject_cast<const QMainWindow*>(dock->parentWidget()) : nullptr;
133 if (!window)
134 return std::numeric_limits<int>::max();
135 int beside = window->width();
136 for (QDockWidget* other : window->findChildren<QDockWidget*>(QString(), Qt::FindDirectChildrenOnly)) {
137 if (other != dock && other->isVisible() && !other->isFloating() && window->dockWidgetArea(other) == Qt::LeftDockWidgetArea)
138 beside -= other->width();
139 }
140 return beside * 5 / 8;
141 }
