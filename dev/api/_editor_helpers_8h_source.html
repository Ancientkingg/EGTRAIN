63inline QString diagnosticsDialogText(const std::vector<SceneDiagnostic>& diagnostics) {
64 const std::size_t kShown = 3;
65 std::vector<const SceneDiagnostic*> listed;
66 for (const SceneDiagnostic& d : diagnostics) {
67 if (d.severity == SceneSeverity::Error)
68 listed.push_back(&d);
69 }
70 const bool errors = !listed.empty();
71 if (!errors) {
72 for (const SceneDiagnostic& d : diagnostics)
73 listed.push_back(&d);
74 }
75 QStringList parts;
76 for (std::size_t index = 0; index < std::min(listed.size(), kShown); ++index) {
77 const SceneDiagnostic& d = *listed[index];
78 QString part = QString::fromStdString(toDialogText(d));
79 if (!d.file.empty() && d.message.find(d.file) == std::string::npos)
80 part += "\nFile: " + display_path::format(QString::fromStdString(d.file));
81 if (!d.path.empty())
82 part += "\nPath: " + QString::fromStdString(d.path);
83 parts << part;
84 }
85 if (listed.size() > kShown) {
86 const long long rest = static_cast<long long>(listed.size() - kShown);
87 parts << (errors ? countText(rest, "more error", "more errors") : countText(rest, "more message", "more messages"));
88 }
89 return parts.join("\n\n");
90}
