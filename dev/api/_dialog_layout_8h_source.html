32inline int heightForContent(QDialog& dialog, int width) {
33 // The style sheet adds padding and borders to labels when they are polished, which changes their size hints.
34 dialog.ensurePolished();
35 QLayout* layout = dialog.layout();
36 auto* scroll = dialog.findChild<QScrollArea*>(QStringLiteral("dialogBodyScroll"));
37 if (!layout || !scroll)
38 return 0;
39 const QMargins margins = layout->contentsMargins();
40 const int innerWidth = width - margins.left() - margins.right();
41 const int frame = 2 * scroll->frameWidth();
42 // A vertical scroll bar that shows stays until the content fits at the width beside it, so the content is measured
43 // at that width.
44 const int scrollBar = scroll->verticalScrollBar()->isVisible() ? scroll->width() - frame - scroll->viewport()->width() : 0;
45 int height = margins.top() + margins.bottom();
46 int shown = 0;
47 for (int i = 0; i < layout->count(); ++i) {
48 QLayoutItem* item = layout->itemAt(i);
49 if (item->isEmpty())
50 continue;
51 ++shown;
52 if (item->widget() == scroll) {
53 QWidget* content = scroll->widget();
54 height += frame + (content->hasHeightForWidth() ? content->heightForWidth(innerWidth - frame - scrollBar) : content->sizeHint().height());
55 } else {
56 height += item->hasHeightForWidth() ? item->heightForWidth(innerWidth) : item->sizeHint().height();
57 }
58 }
59 return height + qMax(0, shown - 1) * layout->spacing();
60}

75inline QScrollArea* install(QDialog& dialog, const QString& heading,
76 const QString& context, QWidget* body,
77 QDialogButtonBox* actions,
78 QRect availableGeometry = QRect()) {
79 if (!availableGeometry.isValid()) {
80 QScreen* screen = QGuiApplication::screenAt(dialog.mapToGlobal(QPoint(0, 0)));
81 if (!screen)
82 screen = QGuiApplication::primaryScreen();
83 if (screen)

87 dialog.setProperty("dialogPresentation", true);

90 const int spacing = qMax(8, lineHeight / 2);
91 auto* layout = new QVBoxLayout(&dialog);
92 layout->setContentsMargins(margin, margin, margin, margin);
93 layout->setSpacing(spacing);
94
95 dialog.setWindowTitle(heading);
96
97 auto* content = new QWidget;
98 auto* contentLayout = new QVBoxLayout(content);
99 contentLayout->setContentsMargins(0, 0, 0, 0);
100 contentLayout->setSpacing(spacing);
101 if (!context.isEmpty()) {
102 auto* subtitle = new QLabel(context, content);
103 subtitle->setObjectName(QStringLiteral("dialogContext"));
104 subtitle->setTextFormat(Qt::PlainText);
105 subtitle->setWordWrap(true);
106 subtitle->setMinimumWidth(0);
107 subtitle->setSizePolicy(QSizePolicy::Ignored, QSizePolicy::Preferred);
108 contentLayout->addWidget(subtitle);
109 }
110 // The body takes the height that the content has left over, so that the context line keeps its own height.
111 contentLayout->addWidget(body, 1);
112 FocusWheel::guardAll(body);
113
114 auto* scroll = new QScrollArea(&dialog);
115 scroll->setObjectName(QStringLiteral("dialogBodyScroll"));
116 scroll->setWidgetResizable(true);
117 scroll->setFrameShape(QFrame::NoFrame);
118 scroll->setMinimumSize(0, 0);
119 // The scroll area draws no focus, so it is no Tab stop, and a click on the body leaves the focus where it is.
120 scroll->setFocusPolicy(Qt::NoFocus);
121 scroll->setWidget(content);
122 QObject::connect(qApp, &QApplication::focusChanged, scroll,
123 [scroll, content](QWidget*, QWidget* focused) {
124 if (focused && content->isAncestorOf(focused))
125 scroll->ensureWidgetVisible(focused);
126 });
127 layout->addWidget(scroll, 1);
128 layout->addWidget(actions);
129 // Tab enters the footer at the first button of its layout, to which the button box passes the focus, and goes on in
130 // the order the buttons were added, which can put a button before that one and out of reach. The buttons follow each
131 // other in the order of the layout, so that Tab and Shift+Tab visit each one once.
132 QWidget* previous = nullptr;
133 for (int i = 0; i < actions->layout()->count(); ++i) {
134 if (QWidget* button = actions->layout()->itemAt(i)->widget()) {
135 if (previous)
136 QWidget::setTabOrder(previous, button);
137 previous = button;
138 }
139 }
140 // In a footer laid out for macOS a footer button that gets the focus does not take the default, as in native dialogs,
141 // so Return still presses the default button. Elsewhere it does, and Return presses that button.
142 const bool macFooter = actions->style()->styleHint(QStyle::SH_DialogButtonLayout, nullptr, actions) == QDialogButtonBox::MacLayout;
143 // The button that is the default when the dialog opens: the one the caller chose, else the first accept button,
144 // which the button box makes the default when it is shown.
145 QPushButton* defaultButton = nullptr;
146 for (QAbstractButton* button : actions->buttons()) {
147 auto* push = qobject_cast<QPushButton*>(button);
148 if (push && macFooter)
149 push->setAutoDefault(false);
150 if (push && (push->isDefault() || (!defaultButton && actions->buttonRole(push) == QDialogButtonBox::AcceptRole)))
151 defaultButton = push;
152 button->style()->unpolish(button);
153 button->style()->polish(button);
154 }
155 // A button made the default before its button box was in the dialog is not the default of the dialog yet. The dialog
156 // would then make the first auto default push button its own default when it opens, and give the default to that
157 // button whenever a button that took it loses the focus. Made the default again here, the chosen button is the one
158 // the dialog keeps.
159 if (defaultButton && defaultButton->isDefault()) {
160 defaultButton->setDefault(false);
161 defaultButton->setDefault(true);
162 }
163 // The first field of the body takes the focus, else the default button. Otherwise Qt gives the focus to the button
164 // box, which comes first in the focus chain and passes it to the first button of its layout, Cancel on macOS, and
165 // that button takes the default while it has the focus. An item view is not taken, because one without a current
166 // row makes its first row current when it gets the focus.
167 QWidget* focus = defaultButton;
168 for (QWidget* widget = body->nextInFocusChain(); body->isAncestorOf(widget); widget = widget->nextInFocusChain()) {
169 if ((widget->focusPolicy() & Qt::TabFocus) && widget->isEnabled() && widget->isVisibleTo(&dialog)
170 && !qobject_cast<QPushButton*>(widget) && !qobject_cast<QToolButton*>(widget) && !qobject_cast<QAbstractItemView*>(widget)) {
171 focus = widget;
172 break;
173 }
174 }
175 if (focus)
176 focus->setFocus();
177
178 // The scroll viewport, not its content, determines the dialog's minimum size.
179 // Reserve room for window decorations and keep the footer outside the viewport.
180 if (availableGeometry.isValid()) {
181 const int maxWidth = qMax(1, availableGeometry.width() * 9 / 10);
182 const int maxHeight = qMax(1, availableGeometry.height() * 4 / 5);
183 dialog.setMaximumSize(maxWidth, maxHeight);
184 const int width = qMin(maxWidth, qMax(480, lineHeight * 32));
185 dialog.resize(width, qMin(maxHeight, detail::heightForContent(dialog, width)));
186 }
187 return scroll;
188}

194inline void fitWidthToContent(QDialog& dialog) {
195 auto* scroll = dialog.findChild<QScrollArea*>(QStringLiteral("dialogBodyScroll"));
196 if (!scroll || !dialog.layout())
197 return;
198 // The layouts of the body keep size hints from before the body was filled or changed, also in a dialog that is not
199 // shown yet, so they are brought up to date before the body is measured.
200 for (QLayout* layout : scroll->widget()->findChildren<QLayout*>())
201 layout->activate();
202 const QMargins margins = dialog.layout()->contentsMargins();
203 const int needed = margins.left() + margins.right() + 2 * scroll->frameWidth()
204 + scroll->widget()->sizeHint().width() + scroll->verticalScrollBar()->sizeHint().width();
205 const int width = qMin(dialog.maximumWidth(), qMax(dialog.width(), needed));
206 dialog.resize(width, qMin(dialog.maximumHeight(), qMax(dialog.height(), detail::heightForContent(dialog, width))));
207}

218inline void showMessage(QLabel* label, const QString& text, const QVariant& status, const QVariant& help) {
219 label->setTextFormat(Qt::PlainText);
220 label->setWordWrap(true);
221 QSizePolicy policy = label->sizePolicy();
222 policy.setHorizontalPolicy(QSizePolicy::Ignored);
223 label->setSizePolicy(policy);
224 label->setText(text);

226 label->setProperty("dialogHelp", help);
227 label->style()->unpolish(label);
228 label->style()->polish(label);
229 label->setVisible(!text.isEmpty());
230 // A message in a shown dialog adds height to the body. Once the layout has made room for it, the dialog grows
231 // to fit, up to its height limit, so that the body does not scroll the message into view.
232 auto* dialog = qobject_cast<QDialog*>(label->window());
233 if (!text.isEmpty() && dialog && dialog->isVisible() && dialog->findChild<QScrollArea*>(QStringLiteral("dialogBodyScroll")))
234 QTimer::singleShot(0, dialog, [dialog] { fitWidthToContent(*dialog); });
235}
