153// A train that has braked into a stop is parked kStopHoldbackM short of the stopping point; the
154// parking assignments below write the same 0.0001 as a literal. The parked position and a
155// stopping point that is compared with it are rounded differently on targets that fuse multiply
156// and add, so they can differ by a few units in the last place in either direction. The tolerance
157// is far above that error and far below any real distance. A position within the tolerance of the
158// hold-back position is at the stop: it is neither short of it nor past it.
159inline constexpr double kStopHoldbackM = 0.0001;
160inline constexpr double kStopHoldbackToleranceM = 1e-6;
161inline bool isShortOfBrakingPoint(double position, double brakingPoint) {
162 return position < brakingPoint - kStopHoldbackM - kStopHoldbackToleranceM;
163}
164inline bool isPastStopHoldback(double position, double stoppingPoint) {
165 return position > stoppingPoint - kStopHoldbackM + kStopHoldbackToleranceM;
166}
