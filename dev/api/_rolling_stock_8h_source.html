171// A train that has braked into a stop is parked kStopHoldbackM short of the stopping point, as the
172// standing-train steps of RollingStockTrajectory.cpp park it. The parked position and a
173// stopping point that is compared with it are rounded differently on targets that fuse multiply
174// and add, so they can differ by a few units in the last place in either direction. The tolerance
175// is far above that error and far below any real distance. A position within the tolerance of the
176// hold-back position is at the stop: it is neither short of it nor past it.
177inline constexpr double kStopHoldbackM = 0.0001;
178inline constexpr double kStopHoldbackToleranceM = 1e-6;
180inline bool isShortOfBrakingPoint(double position, double brakingPoint) {
181 return position < brakingPoint - kStopHoldbackM - kStopHoldbackToleranceM;
182}
183
184inline bool isPastStopHoldback(double position, double stoppingPoint) {
185 return position > stoppingPoint - kStopHoldbackM + kStopHoldbackToleranceM;
186}
