167// A train that has braked into a stop is parked kStopHoldbackM short of the stopping point; the
168// parking assignments below write the same 0.0001 as a literal. The parked position and a
169// stopping point that is compared with it are rounded differently on targets that fuse multiply
170// and add, so they can differ by a few units in the last place in either direction. The tolerance
171// is far above that error and far below any real distance. A position within the tolerance of the
172// hold-back position is at the stop: it is neither short of it nor past it.
173inline constexpr double kStopHoldbackM = 0.0001;
174inline constexpr double kStopHoldbackToleranceM = 1e-6;
176inline bool isShortOfBrakingPoint(double position, double brakingPoint) {
177 return position < brakingPoint - kStopHoldbackM - kStopHoldbackToleranceM;
178}
179
180inline bool isPastStopHoldback(double position, double stoppingPoint) {
181 return position > stoppingPoint - kStopHoldbackM + kStopHoldbackToleranceM;
182}
