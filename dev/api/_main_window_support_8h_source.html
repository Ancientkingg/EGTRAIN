34// Seconds since midnight of an "hh:mm:ss" base time; 0 when the text is empty or not a time of day.
35inline int baseTimeToSeconds(const std::string& hhmmss) {
36 if (hhmmss.empty())
37 return 0;
38 int h = 0, m = 0, s = 0;
39 char extra = 0;
40 if (std::sscanf(hhmmss.c_str(), "%d:%d:%d%c", &h, &m, &s, &extra) != 3)
41 return 0;
42 if (h < 0 || h > 23 || m < 0 || m > 59 || s < 0 || s > 59)
43 return 0;
44 return h * 3600 + m * 60 + s;
45}

54// What the train list reports while 'frames' shows frames that change no state. Every row gets a
55// marker text first. The list is left alone only if it reports no change and each marker is still
56// there, so a row that is written again with the text it already has is noticed too.
57struct FollowListUnderFrames {
58 int changes = 0;
59 bool rowsKept = true;
60};

62inline FollowListUnderFrames showFramesOverMarkedList(QComboBox* list, const std::function<void()>& frames) {
63 const QString marker = QStringLiteral("not written");
64 for (int row = 0; row < list->count(); ++row)
65 list->setItemText(row, marker);
66 FollowListUnderFrames result;
67 const auto countChange = [&result]() { ++result.changes; };
68 const QAbstractItemModel* model = list->model();
69 const QList<QMetaObject::Connection> counters{
70 QObject::connect(model, &QAbstractItemModel::dataChanged, list, countChange),
71 QObject::connect(model, &QAbstractItemModel::rowsInserted, list, countChange),
72 QObject::connect(model, &QAbstractItemModel::rowsRemoved, list, countChange),
73 QObject::connect(model, &QAbstractItemModel::modelReset, list, countChange)};
74 frames();
75 for (const QMetaObject::Connection& counter : counters)
76 QObject::disconnect(counter);
77 for (int row = 0; row < list->count(); ++row)
78 result.rowsKept = result.rowsKept && list->itemText(row) == marker;
79 return result;
80}
