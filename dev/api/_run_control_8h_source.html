24// The stop, pause and step delay of one run, and a bounded queue of commands for them. Every run has
25// one, also a run without a window. No Qt.
26//
27// The request functions, the readers and submit can be called from any thread. beginStep and
28// applyQueuedCommands are called by the thread that runs the steps only.
29//
30// A command waits in the queue until the run reaches the top of a step. There beginStep takes every
31// waiting command and applies it in the order of the submits, then checks pause, stop and the delay.
32// No command takes effect in the middle of a step.
33class RunControl {
34public:
35 // Number of commands that can wait for the next step.
36 static constexpr std::size_t kCommandQueueSize = 16;
37
38 // Ends the run at the next step. The stop is also recorded as a cancellation, and lifts a pause so
39 // that the thread of the run can see the stop.
40 void requestStop();
41 void requestPause();
42 void requestResume();
43 void setDelayMs(int ms);
44
45 bool isStopRequested() const { return m_stop; }
46 // Observation-only latch; set by requestStop() and never cleared.
47 bool wasCancellationRequested() const { return m_cancelled; }
48 bool isPauseRequested() const { return m_pause; }
49 int delayMs() const { return m_delayMs; }
50
51 // Queues a command for the top of the next step. Returns false and queues nothing when
52 // kCommandQueueSize commands wait. Never waits for the run.
53 bool submit(const RunCommand& command);
54
55 // Applies the waiting commands in the order in which they were submitted.
56 void applyQueuedCommands();
57
58 // The top of a step. Applies the waiting commands, waits while the run is paused (and applies the
59 // commands that arrive meanwhile), then sleeps for the step delay. Returns false when the run is
60 // to stop; the delay is not waited for then.
61 //
62 // fetchCommands, when it is given, is called before the waiting commands are applied, also after
63 // each wait of a pause. It lets the owner of the run submit commands that come from another
64 // source, which a paused run could not otherwise take.
65 bool beginStep(const std::function<void()>& fetchCommands = {});
66
67private:
68 std::atomic<bool> m_stop{false};
69 std::atomic<bool> m_cancelled{false};
70 std::atomic<bool> m_pause{false};
71 std::atomic<int> m_delayMs{0};
72
73 // m_queued is the size of m_commands, so a step without commands takes no lock.
74 std::atomic<std::size_t> m_queued{0};
75 std::mutex m_commandsMutex;
76 std::vector<RunCommand> m_commands;
77};
