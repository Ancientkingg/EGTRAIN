21// A facet over one list of the model whose items have no id and name sections instead, such as the dependencies between
22// blocks. A section is chosen in a combo of the sections of the scene, which writes the model as soon as another
23// section is chosen. The other cells are text. A new item is empty, and a row is named by its first section when it is
24// deleted. Nothing else refers to these items.

62 std::vector<Item>& items = panel.mutableModel().*m_list;
63 const int row = panel.table()->currentRow();
64 if (row < 0 || row >= static_cast<int>(items.size()))
65 return;
66 const std::string title = rowTitle(items[static_cast<std::size_t>(row)]);
67 const QString name = title.empty() ? QStringLiteral("incomplete row") : QString::fromStdString(title);
68 if (!panel.confirmDelete(name, {}))
69 return;
70 items.erase(items.begin() + row);
71 panel.setSelectionId(QString());
