9// Whether the selected train can be followed, and the words that say why not.
10// The decision uses only what the window knows, so it has no Qt dependency.
11//
12// The window asks this unit whether the control is offered (controlEnabled), whether Follow
13// can be switched on for the train (canArm), whether it has to be switched off (switchOff),
14// whether the view and the station emphasis follow the train (canAct), whether the view
15// glides to the train or cuts to it (glide) and which text is shown (entryText, statusText).
16// The sentences are true whether Follow is on or off. They say the state of the train first
17// and the part about Follow last, so that a status area that is too narrow elides the part
18// about Follow first.
19
20enum class FollowPhase {
21 NoServices, // the case defines no services
22 NoRun, // the case has services but has not been run yet
23 NotEntered, // the train has not entered the network at the displayed time
24 Running, // the train is in the network and can be drawn
25 Finished, // the train has left the network
26 HiddenByLayer, // the train is in the network but the Trains layer is off
27 NotDrawable // the train is in the network but has no position on the map
28};

73inline FollowPhase followPhase(const FollowAvailabilityInput& in) {
74 if (!in.sceneHasServices)
75 return FollowPhase::NoServices;
76 if (!in.hasRun)
77 return FollowPhase::NoRun;
78 if (!in.train)
79 return FollowPhase::NotEntered;
80 if (in.train->outOfSimulation)
81 return FollowPhase::Finished;
82 if (!guiReplayTrainHasPosition(*in.train, in.timestep))
83 return FollowPhase::NotEntered;
84 if (!in.trainsLayerVisible)
85 return FollowPhase::HiddenByLayer;
86 return in.drawable ? FollowPhase::Running : FollowPhase::NotDrawable;
87}

89inline FollowAvailability followAvailability(const FollowAvailabilityInput& in) {
90 FollowAvailability out;
91 out.phase = followPhase(in);
92 const std::string& name = in.name;
93 switch (out.phase) {
94 case FollowPhase::NoServices:
95 out.entryText = "No trains to follow";
96 out.statusText = "This case has no trains to follow.";
97 break;
98 case FollowPhase::NoRun:
99 out.entryText = "Run the case to follow trains";
100 out.statusText = "Run the case first, then you can follow a train.";
101 break;
102 case FollowPhase::NotEntered: {
103 out.canArm = true;
104 // A train that is scheduled after the end of a completed run has no position at any time of its replay, so the
105 // sentence does not say that Follow starts.
106 const bool afterRun = in.train && in.replayEndTime >= 0 && in.train->departureTime > in.replayEndTime;
107 const std::string tail = in.followOn ? "Follow starts when it enters." : "Follow can be switched on now and starts when it enters.";
108 if (in.train) {
109 const std::string entry = formatSimTime(in.train->departureTime, in.clockOffsetSeconds);
110 out.entryText = name + " (scheduled " + entry + ")";
111 out.statusText = name + " is scheduled to enter at " + entry
112 + (afterRun ? std::string(", after the end of this run, so the view cannot follow it.") : ". " + tail);
113 } else {
114 out.entryText = name + " (not entered yet)";
115 out.statusText = name + " has not entered the network yet. " + tail;
116 }
117 break;
118 }
119 case FollowPhase::Running:
120 out.canArm = true;
121 out.entryText = name + " (running)";
122 out.statusText = name + " is running. " + (in.followOn ? "The view follows it." : "Follow can be switched on now.");
123 break;
124 case FollowPhase::Finished:
125 out.entryText = name + " (finished)";
126 if (in.replay) {
127 out.canArm = true;
128 out.statusText = name + " has left the network at this time. "
129 + (in.followOn ? "Follow stays on and continues when you go back in the replay."
130 : "Follow can be switched on now and continues when you go back in the replay.");
131 } else {
132 out.statusText = name + " has left the network. " + (in.followOn ? "Follow is switched off." : "Follow cannot be switched on for it.");
133 }
134 break;
135 case FollowPhase::HiddenByLayer:
136 out.canArm = true;
137 out.entryText = name + " (hidden)";
138 out.statusText = name + " is running, but the Trains layer is switched off"
139 + (in.followOn ? ", so the view does not follow it."
140 : ". Follow can be switched on now and moves the view when the layer is switched on.");
141 break;
142 case FollowPhase::NotDrawable:
143 out.canArm = true;
144 out.entryText = name + " (no position)";
145 out.statusText = name + " is running, but it has no position on the map to follow.";
146 break;
147 }
148 out.controlEnabled = out.phase != FollowPhase::NoServices && out.phase != FollowPhase::NoRun;
149 out.switchOff = in.followOn && !out.canArm;
150 out.canAct = in.followOn && out.phase == FollowPhase::Running;
151 out.glide = out.canAct && !in.replay;
152 return out;
153}
